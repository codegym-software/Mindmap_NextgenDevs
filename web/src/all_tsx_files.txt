// Decode JWT (Ã„â€˜Ã¡Â»Æ’ lÃ¡ÂºÂ¥y thÃƒÂ´ng tin user tÃ¡Â»Â« id_token)
function decodeJwt(token: string) {
 try {
  const base64Url = token.split(".")[1];
  const base64 = base64Url.replace(/-/g, "+").replace(/_/g, "/");
  const jsonPayload = decodeURIComponent(
   atob(base64)
    .split("")
    .map(c => "%" + ("00" + c.charCodeAt(0).toString(16)).slice(-2))
    .join("")
  );
  return JSON.parse(jsonPayload);
 } catch {
  return null;
 }
}

import { create } from "zustand";
import { isEqual } from "lodash"; // Cáº§n cÃ i `lodash` vÃ  `@types/lodash`

export type NodeData = { id: string; text: string; x: number; y: number; color?: string; parentId?: string; side?: "left" | "right"; collapsed?: boolean; };
export type EdgeData = { id: string; from: string; to: string };

type Snapshot = { nodes: NodeData[]; edges: EdgeData[] };

const MAX_HISTORY = 100;

type State = {
  nodes: NodeData[];
  edges: EdgeData[];
  history: Snapshot[];
  future: Snapshot[];
  setGraph: (n: NodeData[], e: EdgeData[]) => void;
  push: (n: NodeData[], e: EdgeData[]) => void;
  undo: () => Snapshot | null;
  redo: () => Snapshot | null;
  clear: () => void;
};

export const useEditorStore = create<State>((set, get) => ({
  nodes: [],
  edges: [],
  history: [],
  future: [],

  setGraph: (n, e) => {
    set({ nodes: n, edges: e });
  },

  push: (n, e) => {
    const currentState: Snapshot = { nodes: n, edges: e };
    const lastHistoryState = get().history.at(-1);

    // FIX: Only push to history if the state has actually changed
    if (!lastHistoryState || !isEqual(lastHistoryState, currentState)) {
      const nextHistory = [...get().history, currentState].slice(-MAX_HISTORY);
      set({ history: nextHistory, future: [] });
    }
  },

  undo: () => {
    const h = get().history.slice();
    if (h.length <= 1) return null; // Keep the initial state

    const current = h.pop()!;
    const prev = h.at(-1)!;

    set({ 
        history: h, 
        future: [current, ...get().future], 
        nodes: prev.nodes, 
        edges: prev.edges 
    });
    return prev;
  },

  redo: () => {
    const f = get().future.slice();
    if (!f.length) return null;

    const next = f.shift()!;
    set({ 
        history: [...get().history, next], 
        future: f, 
        nodes: next.nodes, 
        edges: next.edges 
    });
    return next;
  },

  clear: () => set({ history: [], future: [] }),
}));

import { cognitoConfig } from './cognitoConfig';

export function getLogoutUrl(): string {
 const u = new URL(`https://${cognitoConfig.Domain}/logout`);
 u.searchParams.set("client_id", cognitoConfig.ClientId);
 u.searchParams.set("logout_uri", cognitoConfig.RedirectUri.replace('/callback', '/dashboard')); // Redirect to dashboard after logout
 return u.toString();
}

export async function exchangeCodeForTokens(code: string) {
  const body = new URLSearchParams({
    grant_type: 'authorization_code',
    client_id: cognitoConfig.ClientId,
    code,
    redirect_uri: cognitoConfig.RedirectUri,
  });

  const res = await fetch(`https://${cognitoConfig.Domain}/oauth2/token`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
    body,
  });

  if (!res.ok) {
    const errorBody = await res.json();
    console.error("Token exchange failed:", errorBody);
    throw new Error(`token_exchange_failed: ${errorBody.error_description || res.status}`);
  }

  return res.json() as Promise<{
    access_token: string;
    id_token: string;
    refresh_token?: string;
    expires_in?: number;
  }>;
}

      if (err) {
        return reject(err);

      if (session && session.isValid()) {

import { useCallback } from "react";
import { useMindmapsStore } from "../app/store/useMindmapsStore";

const KEY = "mm_guest_maps";          // danh sÃ¡ch guest (dashboard)
const BUCKET = "mm_guest_docs";       // ná»™i dung mindmap guest (editor)

export type GuestItem = { id: string; name: string; createdAt: string };

export function useLocalMindmap() {
  const set = useMindmapsStore((s: { set: any }) => s.set);

  // Load danh sÃ¡ch guest vÃ o store
  const load = useCallback(() => {
    const raw = localStorage.getItem(KEY);
    const arr: GuestItem[] = raw ? JSON.parse(raw) : [];
    set({ items: arr });
  }, [set]);

  // List táº¥t cáº£ guest items
  const listGuests = useCallback((): GuestItem[] => {
    try { 
      return JSON.parse(localStorage.getItem(KEY) || "[]"); 
    } catch { 
      return []; 
    }
  }, []);

  // Táº¡o guest má»›i
  const createGuest = useCallback(() => {
    const arr: GuestItem[] = listGuests();
    const id = "guest-" + Date.now();
    const item: GuestItem = { 
      id, 
      name: "Mindmap má»›i", 
      createdAt: new Date().toISOString() 
    };
    const next = [item, ...arr].slice(0, 200); // TÄƒng limit lÃªn 200
    localStorage.setItem(KEY, JSON.stringify(next));
    set({ items: next });

    // Khá»Ÿi táº¡o doc rá»—ng trong BUCKET
    const docs = JSON.parse(localStorage.getItem(BUCKET) || "{}");
    if (!docs[id]) {
      docs[id] = { 
        id, 
        name: item.name, 
        content: { 
          nodes: { root: { id: "root", text: "Root", x: 0, y: 0 } }, 
          edges: [] 
        } 
      };
      localStorage.setItem(BUCKET, JSON.stringify(docs));
    }

    return item;
  }, [listGuests, set]);

  // Cáº­p nháº­t tÃªn guest
  const updateGuestName = useCallback((id: string, name: string) => {
    const arr: GuestItem[] = listGuests();
    const next = arr.map((x) => (x.id === id ? { ...x, name } : x));
    localStorage.setItem(KEY, JSON.stringify(next));
    set({ items: next });
  }, [listGuests, set]);

  // XÃ³a guest
  const removeGuest = useCallback((id: string) => {
    // XÃ³a trong danh sÃ¡ch
    const arr: GuestItem[] = listGuests();
    const next = arr.filter((x) => x.id !== id);
    localStorage.setItem(KEY, JSON.stringify(next));
    set({ items: next });

    // XÃ³a ná»™i dung trong bucket
    const docs = JSON.parse(localStorage.getItem(BUCKET) || "{}");
    if (docs[id]) {
      delete docs[id];
      localStorage.setItem(BUCKET, JSON.stringify(docs));
    }
  }, [listGuests, set]);

  return { 
    load, 
    createGuest, 
    updateGuestName, 
    removeGuest, 
    listGuests 
  };
}

  const hasRun = useRef(false);

      const url = new URL(window.location.href);
      const code = url.searchParams.get("code");
      const error = url.searchParams.get("error");

      // Handle OAuth errors from Cognito

        console.error("OAuth error:", error, url.searchParams.get("error_description"));
        navigate("/dashboard", { replace: true });
        return;

      // Check if code is present

        console.warn("Callback skipped: No authorization code found.");
        navigate("/dashboard", { replace: true });
        return;
      }

      try {
        const res = await exchangeCodeForTokens(code);
        
        // Save the tokens to global state and local storage
        setAuthTokens({
          access_token: res.access_token,
          id_token: res.id_token,
          refresh_token: res.refresh_token,
          expires_at: Math.floor(Date.now() / 1000) + (res.expires_in ?? 3600),
        });

        // Trigger storage event for other tabs to sync
        window.dispatchEvent(new StorageEvent("storage", { key: "mm_tokens" }));
        
        // Navigate to the dashboard
        navigate("/dashboard", { replace: true });

      } catch (e) {
        console.error("Token exchange error:", e instanceof Error ? e.message : e);
        navigate("/dashboard", { replace: true });

        // Clean up the URL to remove the code and state parameters

              nodes: Object.fromEntries(currentNodes.map(n => [n.id, n])),
              edges: currentEdges,
          },
      };

      if (isGuest) {
          saveGuestDoc(id, { id, ...docToSave });

              console.error("Auto-save failed:", e);
              addToast("LÆ°u tá»± Ä‘á»™ng tháº¥t báº¡i", "error");
          });
      }
    }, 1500); // 1.5 second debounce

	return () => {
      clearTimeout(handler);
    };
  }, [nodes, edges, name, isDataLoaded, id, isGuest, isAuthed, addToast]);

  const handleSave = () => {
    if (!isAuthed) {
        addToast("Vui lÃ²ng Ä‘Äƒng nháº­p Ä‘á»ƒ lÆ°u mindmap.", "info");
        login();
    } else if (id) {
        const content = { nodes: Object.fromEntries(nodes.map(n => [n.id, n])), edges };
        mindmapsApi.update(id, { name, content })
            .then(() => addToast("ÄÃ£ lÆ°u mindmap!", "success"))
            .catch(e => {
                console.error("Save failed:", e);
                addToast("LÆ°u tháº¥t báº¡i", "error");

  // ================================================
  // START: YÃŠU Cáº¦U Cáº¬P NHáº¬T 1 (NODE Gá»C VÃ€ EDITING)

  const getNodeBox = useCallback((nodeText: string, isRoot: boolean = false) => {
    // 1a. TÃ­nh toÃ¡n style Ä‘á»™ng dá»±a trÃªn isRoot
    const finalFontSize = isRoot ? FONT_SIZE + 4 : FONT_SIZE; // Chá»¯ to hÆ¡n
    const finalLineHeight = finalFontSize * LINE_HEIGHT_MULTIPLIER;
    const nodeWidthChars = isRoot ? NODE_WIDTH_CHARS + 5 : NODE_WIDTH_CHARS;
    const approxCharWidth = isRoot ? APPROX_CHAR_WIDTH + 1 : APPROX_CHAR_WIDTH;
    const minH = finalFontSize * LINE_HEIGHT_MULTIPLIER + PADDING_Y * 2;
    const minW = (nodeWidthChars * approxCharWidth) + PADDING_X * 2; // Rá»™ng hÆ¡n

    const lines = (nodeText || "").split("\n");
    const wrappedLines: string[] = [];

    lines.forEach(line => {
      if (line.length <= nodeWidthChars) {
        wrappedLines.push(line);
      } else {
        // Simple word wrap
        let currentLine = "";
        const words = line.split(' ');
        for (const word of words) {
          if (word.length > nodeWidthChars) {
            if (currentLine) wrappedLines.push(currentLine);
            wrappedLines.push(word);
            currentLine = "";
          } else if ((currentLine + " " + word).length > nodeWidthChars) {
            wrappedLines.push(currentLine);
            currentLine = word;
          } else {
            currentLine += (currentLine ? " " : "") + word;
          }
        }
        if (currentLine) wrappedLines.push(currentLine);
      }
    });

    const w = minW; // Chiá»u rá»™ng cá»‘ Ä‘á»‹nh
    const h = Math.max(minH, wrappedLines.length * finalLineHeight + PADDING_Y * 2);
    
    // Tráº£ vá» text Ä‘Ã£ Ä‘Æ°á»£c ngáº¯t dÃ²ng VÃ€ FONT_SIZE Ä‘á»ƒ render
	return { w, h, textToRender: wrappedLines.join("\n"), finalFontSize };
  }, []);
  
  const handleUndo = useCallback(() => undo(), [undo]);
  const handleRedo = useCallback(() => redo(), [redo]);
  const startEditing = useCallback((nodeId: string) => {

      setGraph(newNodes, edges);
    }
  }, [editingNodeId, nodes, edges, setGraph, pushHistory]);

  const handleAddChild = useCallback((parentId: string) => {
    const parent = nodes.find(n => n.id === parentId);
    if (!parent) return;

    const newId = "n" + Date.now();

    setGraph(newNodes, newEdges);
    startEditing(newId);

  const handleAddSibling = useCallback((nodeId: string) => {
    if (nodeId === 'root') { handleAddChild('root'); return; }
    const parentId = nodes.find(n => n.id === nodeId)?.parentId;
    if (parentId) handleAddChild(parentId);
  }, [nodes, handleAddChild]);

  const handleLayout = useCallback((layoutType: 'TB' | 'LR' = 'TB') => {
    const g = new dagre.graphlib.Graph();
    g.setGraph({ rankdir: layoutType, nodesep: 50, ranksep: 120 });
    g.setDefaultEdgeLabel(() => ({}));

    nodes.forEach(node => {
      // 1a. Láº¥y kÃ­ch thÆ°á»›c node chÃ­nh xÃ¡c (bao gá»“m cáº£ root)
      const { w, h } = getNodeBox(node.text, node.id === 'root');
      g.setNode(node.id, { label: node.text, width: w, height: h });
    });
    edges.forEach(edge => g.setEdge(edge.from, edge.to));

    dagre.layout(g);

    const newNodes = nodes.map(n => {
      const layoutNode = g.node(n.id);
      if (!layoutNode) return n; // Failsafe
      return { ...n, x: layoutNode.x, y: layoutNode.y };
    });

    pushHistory(nodes, edges);
    setGraph(newNodes, edges);
    addToast("ÄÃ£ Ã¡p dá»¥ng layout!", "success");
  }, [nodes, edges, pushHistory, setGraph, getNodeBox, addToast]); // ThÃªm getNodeBox vÃ o dependency

    const findChildren = (id: string) => {
      edges.forEach(e => {
        if (e.from === id) {
          nodesToDelete.add(e.to);
          findChildren(e.to);
        }
      });
    };
    findChildren(nodeId);

    const parentId = nodes.find(n => n.id === nodeId)?.parentId ?? 'root';
    const newNodes = nodes.filter(n => !nodesToDelete.has(n.id));
    const newEdges = edges.filter(e => !nodesToDelete.has(e.from) && !nodesToDelete.has(e.to));
    
    pushHistory(nodes, edges);
    setGraph(newNodes, newEdges);
    setSelectedNodeId(parentId);
  }, [nodes, edges, pushHistory, setGraph]);

  const handleToggleCollapse = useCallback((e: any, nodeId: string) => {

	pushHistory(nodes, edges); // LÆ°u tráº¡ng thÃ¡i trÆ°á»›c khi thay Ä‘á»•i
	const newNodes = nodes.map(n => n.id === nodeId ? { ...n, collapsed: !n.collapsed } : n);
	setGraph(newNodes, edges);
  }, [nodes, edges, pushHistory, setGraph]);

  const handleKeyDown = useCallback((e: KeyboardEvent) => {
    // 1b. KhÃ´ng xá»­ lÃ½ gÃ¬ thÃªm náº¿u Ä‘ang edit (textarea sáº½ tá»± xá»­ lÃ½)
    if (editingNodeId) {
        // (Logic xá»­ lÃ½ Enter/Esc Ä‘Ã£ chuyá»ƒn vÃ o <textarea> bÃªn dÆ°á»›i)
        return;
    }
    if (document.activeElement && ['INPUT', 'TEXTAREA'].includes(document.activeElement.tagName)) return;
    if (e.key === 'Tab') { e.preventDefault(); handleAddChild(selectedNodeId); }
    else if (e.key === 'Enter') { e.preventDefault(); handleAddSibling(selectedNodeId); }
    else if (e.key === 'Delete' || e.key === 'Backspace') { e.preventDefault(); handleDeleteNode(selectedNodeId); }
    else if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'z') { e.preventDefault(); handleUndo(); }
    else if ((e.ctrlKey || e.metaKey) && e.key.toLowerCase() === 'y') { e.preventDefault(); handleRedo(); }
    else if (e.key.length === 1 && /^[a-zA-Z0-9\S]$/.test(e.key)) { e.preventDefault(); startEditing(selectedNodeId); }
  }, [editingNodeId, selectedNodeId, stopEditing, handleAddChild, handleAddSibling, handleDeleteNode, handleUndo, handleRedo, startEditing]);
  
  useEffect(() => {
    window.addEventListener('keydown', handleKeyDown);
	return () => window.removeEventListener('keydown', handleKeyDown);

  const handleWheel = (e: any) => {
    e.evt.preventDefault();
    const scaleBy = 1.05;
    const stage = e.target.getStage();
    const oldScale = stage.scaleX();
    const pointerPosition = stage.getPointerPosition(); // Láº¥y vá»‹ trÃ­ con trá»
    if (!pointerPosition) return; // Kiá»ƒm tra náº¿u khÃ´ng láº¥y Ä‘Æ°á»£c vá»‹ trÃ­
    const mousePointTo = {
      x: (pointerPosition.x - stage.x()) / oldScale,
      y: (pointerPosition.y - stage.y()) / oldScale,
    };
    const newScale = e.evt.deltaY > 0 ? oldScale / scaleBy : oldScale * scaleBy;
    setScale(newScale);
    setPos({
      x: pointerPosition.x - mousePointTo.x * newScale,
      y: pointerPosition.y - mousePointTo.y * newScale,
    });
  };


  // Cáº¬P NHáº¬T: handleDragMove chá»‰ cÃ²n kiá»ƒm tra drop-target
  const handleDragMove = (e: any, draggedNodeId: string) => {
	const pos = e.target.position();
	let targetFound: string | null = null;
	
	// Check for collision with other nodes
	for (const node of nodes) {
		if (node.id === draggedNodeId) continue;
		
        // 1a. Láº¥y kÃ­ch thÆ°á»›c node chÃ­nh xÃ¡c (bao gá»“m cáº£ root)
		const { w, h } = getNodeBox(node.text, node.id === 'root');
		const isOver = (
			pos.x > node.x - w / 2 &&
			pos.x < node.x + w / 2 &&
			pos.y > node.y - h / 2 &&
			pos.y < node.y + h / 2
		);

		if (isOver) {
			targetFound = node.id;
			break;
		}
	}
	setDropTargetId(targetFound);
  };

  // Cáº¬P NHáº¬T: handleDragEnd khÃ´ng cáº§n pushHistory
  const handleDragEnd = (e: any, draggedNodeId: string) => {
	const finalX = e.target.x();
	const finalY = e.target.y();
	
	// Check if we dropped on a valid target
	if (dropTargetId && dropTargetId !== draggedNodeId) {
        const draggedNode = nodes.find(n => n.id === draggedNodeId);
        if (!draggedNode) return; // Should not happen
       
        // Prevent dropping on own child (circular dependency)
        let isDroppingOnChild = false;
        const checkChildren = (id: string) => {
            if (id === dropTargetId) { isDroppingOnChild = true; return; }
            edges.filter(e => e.from === id).forEach(e => checkChildren(e.to));
        };
        checkChildren(draggedNodeId);

        if (isDroppingOnChild) {
            addToast("KhÃ´ng thá»ƒ di chuyá»ƒn node cha vÃ o node con cá»§a nÃ³!", "error");
            // Just do a normal move instead of reparenting
            const newNodes = nodes.map(n => n.id === draggedNodeId ? {...n, x: finalX, y: finalY} : n);
        	setGraph(newNodes, edges);
        	setDropTargetId(null); // Di chuyá»ƒn xuá»‘ng
        	return;
        }

        // --- This is a Reparent Action ---
        const newParentId = dropTargetId;
        const oldEdge = edges.find(e => e.to === draggedNodeId);

        let newEdges: EdgeData[];
        if (oldEdge) {
            // Node already had a parent, just update the 'from'
            newEdges = edges.map(e => e.id === oldEdge.id ? { ...e, from: newParentId } : e);
        } else {
            // Node was a floating topic, create a new edge
            newEdges = [...edges, { id: `e-${draggedNodeId}`, from: newParentId, to: draggedNodeId }];
        }
       
        const newNodes = nodes.map(n => n.id === draggedNodeId ? { ...n, parentId: newParentId, x: finalX, y: finalY } : n);
       
    	setGraph(newNodes, newEdges);
       
	} else {
		// --- This is a Standard Move Action ---
		const newNodes = nodes.map(n => n.id === draggedNodeId ? {...n, x: finalX, y: finalY} : n);
		setGraph(newNodes, edges);
	}
	
	// Reset drop target visual
	setDropTargetId(null);
  };

  // 1b. Cáº­p nháº­t editingNodePosition Ä‘á»ƒ tráº£ vá» style
  const editingNodePosition = useMemo(() => {

  }, [editingNodeId, nodes, scale, pos, getNodeBox]); // ThÃªm getNodeBox vÃ o dependency

    const node = nodeMap.get(nodeId);
    if (!node) return false; // Node khÃ´ng tá»“n táº¡i
	if (nodeId === 'root' || !node.parentId) return true; // LuÃ´n hiá»‡n root vÃ  floating topics

	if (!parent) return true; // Node má»“ cÃ´i (lá»—i data?), cá»© hiá»‡n

    // Sá»¬A Lá»–I TS(2345): ThÃªm kiá»ƒm tra parent.parentId tá»“n táº¡i
	if (!parent.parentId) return true; // Náº¿u cha lÃ  floating topic (khÃ´ng cÃ³ cha ná»¯a), thÃ¬ hiá»‡n

  }, [nodeMap]);

  // END NEW
  
  // Sá»¬A Lá»–I HOOK: TÃ­nh toÃ¡n cÃ¡c node cÃ³ con (hasChildren) á»Ÿ Ä‘Ã¢y, bÃªn ngoÃ i vÃ²ng láº·p
  const nodesWithChildren = useMemo(() => {

  // START: YÃŠU Cáº¦U Cáº¬P NHáº¬T 2 (Äáº¾M CON CHÃU)
  // ================================================
  const descendantCounts = useMemo(() => {
    const counts = new Map<string, number>();
    const adj = new Map<string, string[]>();
    
    // 1. XÃ¢y dá»±ng danh sÃ¡ch ká» (adjacency list)
    for (const edge of edges) {
        if (!adj.has(edge.from)) adj.set(edge.from, []);
        adj.get(edge.from)!.push(edge.to);
    }

    // 2. HÃ m DFS (Äá»‡ quy) Ä‘á»ƒ Ä‘áº¿m con chÃ¡u
    const dfs = (nodeId: string): number => {
        const children = adj.get(nodeId) || [];
        let total = children.length; // Báº¯t Ä‘áº§u báº±ng cÃ¡ch Ä‘áº¿m sá»‘ con trá»±c tiáº¿p
        for (const childId of children) {
            total += dfs(childId); // Cá»™ng thÃªm táº¥t cáº£ con chÃ¡u cá»§a con Ä‘Ã³
        }
        return total;
    };

    // 3. TÃ­nh toÃ¡n cho táº¥t cáº£ cÃ¡c node
    for (const nodeId of nodeMap.keys()) {
        counts.set(nodeId, dfs(nodeId));
    }
    return counts;
  }, [edges, nodeMap]); // Phá»¥ thuá»™c vÃ o edges vÃ  nodeMap (nodeMap phá»¥ thuá»™c vÃ o nodes)
  // ================================================

    			  }}
    			  onMouseUp={() => setIsPanning(false)}

    				if (isPanning) setPos({ x: pos.x + e.evt.movementX, y: pos.y + e.evt.movementY });
    			  }}
    			// NEW: Handle double click on stage to create floating topic
    			onDblClick={(e) => {
    				const stage = e.target.getStage();

    					// Tá»a Ä‘á»™ Ä‘iá»ƒm káº¿t thÃºc (cáº¡nh trÃ¡i cá»§a node con)

    					// Tá»a Ä‘á»™ Ä‘iá»ƒm Ä‘iá»u khiá»ƒn (control points)

    				})}
    				{/* Cáº¬P NHáº¬T: Render visibleNodes */}
    				{visibleNodes.map((node) => {
                        // 1a. Láº¥y style node gá»‘c
                        const isRoot = node.id === 'root';
    					const { w, h, textToRender, finalFontSize } = getNodeBox(node.text, isRoot);
    					const isSelected = node.id === selectedNodeId;
    					const isDropTarget = node.id === dropTargetId;

// src/services/authStorage.ts
export type Tokens = {
  access_token: string;
  refresh_token?: string;
  id_token?: string;
  expires_at?: number; // epoch seconds
};

const KEY = "mm_tokens";

export function loadTokens(): Tokens | null {
  try { return JSON.parse(localStorage.getItem(KEY) || "null"); }
  catch { return null; }
}

export function saveTokens(t: Tokens) {
  localStorage.setItem(KEY, JSON.stringify(t));
}

export function clearTokens() {
  localStorage.removeItem(KEY);
}

export function isExpired(t?: Tokens | null, skewSeconds = 30): boolean {
  if (!t?.expires_at) return true;
  return t.expires_at <= Math.floor(Date.now() / 1000) + skewSeconds;
}
